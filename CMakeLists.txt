  smtg_target_configure_version_file(SonicraftAIStringsQ4)
  target_compile_definitions(SonicraftAIStringsQ4 PUBLIC $<$<CONFIG:Debug>:VSTGUI_LIVE_EDITING=1>)

  endif()
